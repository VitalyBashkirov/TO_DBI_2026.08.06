def _validate_config(self):
    """DS 047: проверка обязательных параметров config."""
    required = {
        'paths': ['logs_dir', 'report_filename'],
    }
    errors = []
    for section, keys in required.items():
        if section not in self.config:
            errors.append(f"Секция [{section}] отсутствует в config.ini")
            continue
        for key in keys:
            if key not in self.config[section]:
                errors.append(f"Параметр [{section}].{key} отсутствует в config.ini")
    
    if errors:
        raise ValueError("Ошибки config.ini:\n  " + "\n  ".join(errors))

def _cleanup_temp_reports(self, max_age_days: int = 7):
    """DS 047: очистка старых отчётов из temp\ при старте (старше 7 дней)."""
    import time
    temp_dir = Path(r'F:\TO_DBI\temp')
    if not temp_dir.exists():
        return
    
    patterns = ['*.md', '*.html', '*.json', '*.tsv', '*.xlsx']
    deleted = 0
    now = time.time()
    cutoff = now - (max_age_days * 86400)
    
    for pattern in patterns:
        for file in temp_dir.glob(pattern):
            # Исключения: тесты, утилиты, скрипты
            if file.name.startswith('test_') or file.name.startswith('run_'):
                continue
            if file.suffix == '.py':
                continue
            try:
                if file.stat().st_mtime < cutoff:
                    file.unlink()
                    deleted += 1
            except Exception as e:
                print(f"[DS 047] Ошибка удаления {file}: {e}")
    
    if deleted > 0:
        print(f"[DS 047] Очищено старых отчётов из temp\\: {deleted}")

## Комментарий от меня

Задача — **гигиена вывода отчётов**. Три связанные правки:

### 1. Фикс вывода

**Проблема**: сканер **пишет** в `temp\` при **отсутствии** `report_filename`.
**Решение**: **принудительно** переносить отчёты в `logs\` **всегда**.

### 2. Валидация config

**Проблема**: `report_filename` **не обязателен** — легко забыть.
**Решение**: `_validate_config()` — **проверка** обязательных параметров при старте.

### 3. Очистка temp

**Проблема**: старые отчёты **накапливаются** в `temp\`.
**Решение**: `_cleanup_temp_reports()` — **автоматически** удалять **старше 7 дней**.

**Исключения**: не удалять **тесты**, **утилиты**, **`.py`** файлы.
