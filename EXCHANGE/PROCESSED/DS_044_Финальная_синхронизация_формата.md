import re

from openpyxl import Workbook

def extract_cft_report(html_path: Path, xlsx_path: Path):
    with open(html_path, 'r', encoding='utf-8') as f:
        html = f.read()
    
    soup = BeautifulSoup(html, 'html.parser')
    
    # Ищем ВСЕ таблицы на странице
    tables = soup.find_all('table')
    print(f"Найдено таблиц: {len(tables)}")
    
    wb = Workbook()
    # Первый лист — удалим (создаётся по умолчанию)
    wb.remove(wb.active)
    
    for idx, table in enumerate(tables, 1):
        ws = wb.create_sheet(title=f'Table_{idx}')
        
        rows = table.find_all('tr')
        for row_idx, row in enumerate(rows, 1):
            cells = row.find_all(['td', 'th'])
            for col_idx, cell in enumerate(cells, 1):
                text = cell.get_text(strip=True)
                ws.cell(row=row_idx, column=col_idx, value=text)
        
        print(f"Table_{idx}: {len(rows)} строк, {len(rows[0].find_all(['td', 'th'])) if rows else 0} колонок")
    
    wb.save(xlsx_path)
    print(f"Сохранено: {xlsx_path}")

"""

import re
from collections import Counter
from pathlib import Path
from openpyxl import load_workbook

def parse_arm_md(path):
    """Парсинг Markdown-отчёта АРМ."""
    rows = []
    with open(path, 'r', encoding='utf-8') as f:
        for line in f:
            # Формат: № CLASS_ID SHORT_NAME SECTION LINE CHECK LEVEL TYPE ERROR PLAN
            m = re.match(r'^\s*\d+\s+(\S+)\s+(\S+)\s+(\S+)\s+(\d+)\s+(\S+)\s+(\S+)\s+(\S+)\s+', line)
            if m:
                rows.append({
                    'class': m.group(1),
                    'short': m.group(2),
                    'section': m.group(3),
                    'line': int(m.group(4)),
                    'check': m.group(5),
                    'level': m.group(6),
                    'type': m.group(7),
                })
    return rows

def parse_cft_xlsx(path):
    """Парсинг XLSX-отчёта ЦФТ."""
    wb = load_workbook(path, read_only=True, data_only=True)
    rows = []
    
    # Ищем лист с CHECK-колонкой
    for sheet_name in wb.sheetnames:
        ws = wb[sheet_name]
        data = list(ws.iter_rows(values_only=True))
        if not data:
            continue
        
        header = [str(c).strip().lower() if c else '' for c in data[0]]
        if 'check' in header:
            check_idx = header.index('check')
            line_idx = header.index('line') if 'line' in header else None
            section_idx = header.index('section') if 'section' in header else None
            type_idx = header.index('type') if 'type' in header else None
            class_idx = header.index('class_id') if 'class_id' in header else None
            short_idx = header.index('short_name') if 'short_name' in header else None
            
            for row in data[1:]:
                if not row or not row[check_idx]:
                    continue
                rows.append({
                    'sheet': sheet_name,
                    'class': row[class_idx] if class_idx is not None else '',
                    'short': row[short_idx] if short_idx is not None else '',
                    'section': row[section_idx] if section_idx is not None else '',
                    'line': row[line_idx] if line_idx is not None else '',
                    'check': row[check_idx],
                    'type': row[type_idx] if type_idx is not None else '',
                })
            break  # берём первый лист с CHECK
    
    wb.close()
    return rows

def _parse_class_and_method(self, file_path, line_number: int = None) -> Tuple[str, str, str]:
    """DS 044: парсинг класса/метода/секции. Для представлений секция = пусто."""
    class_name = "UNKNOWN"
    method_name = "UNKNOWN"
    section = "PRIVATE"
    
    try:
        content, _ = read_file_with_encoding(Path(file_path))
        lines = content.splitlines()
        
        # DS 044: определяем — это представление или операция
        is_view = False
        for line in lines[:50]:
            if re.search(r'^\s*view\s+\w+', line):
                is_view = True
                break
        
        # 1. Класс
        for line in lines:
            class_match = re.search(r'\bclass\s+(\w+)', line)
            if class_match:
                class_name = class_match.group(1)
                break
        
        # 2. Метод / представление
        for line in lines:
            method_match = re.search(r'\bmethod\s+(\w+)\s+is', line, re.IGNORECASE)
            if method_match:
                method_name = method_match.group(1)
                break
            if is_view:
                view_match = re.search(r'^\s*view\s+(\w+)', line)
                if view_match:
                    method_name = view_match.group(1)
                    break
        
        # DS 044: для представлений секция = пусто
        if is_view:
            return class_name, method_name, ""
        
        # 3. Карта секций по строкам
        section_markers = []
        for i, line in enumerate(lines, 1):
            stripped = line.strip().lower()
            m = re.match(r'^(private|execute|validate|public)\s+is\b', stripped)
            if m:
                section_markers.append((i, m.group(1).upper()))
        
        if line_number is not None and section_markers:
            current_section = "PRIVATE"
            for marker_line, marker_section in section_markers:
                if marker_line <= line_number:
                    current_section = marker_section
                else:
                    break
            section = current_section
        elif section_markers:
            section = section_markers[0][1]
    
    except Exception:
        pass
    
    return class_name, method_name, section
