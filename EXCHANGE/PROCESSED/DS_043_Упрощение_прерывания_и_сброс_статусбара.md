ЗАДАЧА B2 — Надёжный перенос файла задания

python
def _move_task_to_processed(task_file_path):
    """DS 043: надёжный перенос файла задания из INBOX в PROCESSED."""
    from pathlib import Path
    import shutil

    inbox_path = Path(task_file_path)
    processed_dir = Path(r"F:\TO_DBI\EXCHANGE\PROCESSED")
    processed_dir.mkdir(parents=True, exist_ok=True)

    if not inbox_path.exists():
        print(f"[DS 043] Файл {inbox_path} не найден в INBOX")
        return False

    target = processed_dir / inbox_path.name
    try:
        shutil.move(str(inbox_path), str(target))
        print(f"[DS 043] Файл перенесён: {inbox_path.name}")
        return True
    except Exception as e:
        print(f"[DS 043] Ошибка переноса {inbox_path.name}: {e}")
        return False
Правило: если хотя бы один шаг протокола провалился (отчёт, лог, перенос) — задача НЕ считается завершённой, файл остаётся в INBOX.

def _verify_task_completion(task_file_name):
    """DS 043: верификация завершения задачи."""
    from pathlib import Path

    checks = {
        "inbox_empty": not (Path(r"F:\TO_DBI\EXCHANGE\INBOX") / task_file_name).exists(),
        "in_processed": (Path(r"F:\TO_DBI\EXCHANGE\PROCESSED") / task_file_name).exists(),
        "report_exists": (Path(r"F:\TO_DBI\EXCHANGE\OUTBOX") / f"{task_file_name}_отчет.md").exists()
                        or (Path(r"F:\TO_DBI\EXCHANGE\OUTBOX") / f"{task_file_name}_отчет.html").exists(),
        "bot_log_updated": _check_bot_log_recent(task_file_name),  # есть запись за последние 5 минут
    }

    failed = [k for k, v in checks.items() if not v]
    if failed:
        print(f"[DS 043] ВНИМАНИЕ: протокол не завершён. Провалено: {failed}")
        return False
    print(f"[DS 043] Протокол завершён: все проверки пройдены")
    return True
