Новый код:

python
def _parse_class_and_method(self, file_path, line_number: int = None) -> Tuple[str, str, str]:
    """
    DS 038: Парсинг имени класса, метода и секции из файла.
    Если line_number указан — секция определяется ПО СТРОКЕ проблемы,
    а не по первому вхождению в файле.
    """
    class_name = "UNKNOWN"
    method_name = "UNKNOWN"
    section = "PRIVATE"
    
    try:
        content, _ = read_file_with_encoding(Path(file_path))
        lines = content.splitlines()
        
        # 1. Класс
        for line in lines:
            class_match = re.search(r'\bclass\s+(\w+)', line)
            if class_match:
                class_name = class_match.group(1)
                break
        
        # 2. Метод
        for line in lines:
            method_match = re.search(r'\bmethod\s+(\w+)\s+is', line, re.IGNORECASE)
            if method_match:
                method_name = method_match.group(1)
                break
        
        # 3. Карта секций по строкам
        # Ищем маркеры секций: private is, execute is, validate is, public is
        section_markers = []  # [(line_num, section_name), ...]
        for i, line in enumerate(lines, 1):
            stripped = line.strip().lower()
            # Ищем "private is", "execute is", "validate is"
            m = re.match(r'^(private|execute|validate|public)\s+is\b', stripped)
            if m:
                section_markers.append((i, m.group(1).upper()))
        
        # DS 038: определяем секцию по номеру строки
        if line_number is not None and section_markers:
            current_section = "PRIVATE"  # по умолчанию до первого маркера
            for marker_line, marker_section in section_markers:
                if marker_line <= line_number:
                    current_section = marker_section
                else:
                    break
            section = current_section
        elif section_markers:
            # Если line_number не указан — берём первую секцию
            section = section_markers[0][1]
    
    except Exception:
        pass
    
    return class_name, method_name, section
