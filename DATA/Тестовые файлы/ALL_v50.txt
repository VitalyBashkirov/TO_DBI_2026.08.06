-- ============================================================================
-- Правило: v50.SQL.OUTERJOIN.п.1.1
-- Описание: Замена (+) на LEFT/RIGHT JOIN
-- Категория: SQL
-- Приоритет: high
-- ============================================================================
-- 3.RUBRICATOR_FIXES.md:
--   Короткое описание: Замена (+) на LEFT/RIGHT JOIN
--   Подробное описание: В Oracle используется оператор (+) для внешнего соединения. 
--   В PostgreSQL и ANSI SQL его нет. Заменить на явные LEFT JOIN или RIGHT JOIN с условием ON.
-- ============================================================================
-- 4.RUBRICATOR_PROMPTS.json:
--   version: 2.2.0
--   search_prompt: Найти в PLPlus коде использование устаревших синтаксисов внешних соединений:
--      1. Оператор Oracle '(+)' в условии WHERE
--      2. Конструкцию '&collection(true)' в условии WHERE
--      3. Конструкцию '[REF](true)' или '[REF](false)' в условии WHERE
--   fix_instruction: Преобразовать устаревший синтаксис в ANSI SQL
--   fix_prompt: Выполни исправление согласно fix_instruction
--   test_generation_prompt: Сгенерируй тестовый PLPlus файл для проверки правила
-- ============================================================================
-- 5.RUBRICATOR_PARSER_SQL.json:
--   version: 1.0.0
--   hybrid_mode: true
--   patterns:
--     - oracle_outer_plus_right: (\w+)\.(\w+)\s*=\s*(\w+)\.(\w+)\(\+\)
--     - oracle_outer_plus_left: (\w+)\.(\w+)\(\+\)\s*=\s*(\w+)\.(\w+)
--     - plplus_ref_param_true: (\w+)%?(\w+)\s*=\s*(\w+)\.\[(\w+)\]\(true\)
--     - plplus_collection_true: (\w+)\.\[(\w+)\]\s*=\s*(\w+)&collection\(true\)
-- ============================================================================

PUBLIC PROCEDURE v50_SQL_OUTERJOIN_p_1_1 IS
    v_ref REF [PRICE_LIST];
    v_result VARCHAR2(4000);
    P_DATE DATE_TIME;
BEGIN
    BEGIN
        savepoint sp1;
        debug_pipe('-= v50.SQL.OUTERJOIN.п.1.1 Замена (+) на LEFT/RIGHT JOIN =-', 0);
        
        -- [-] НЕПРАВИЛЬНО (устаревший синтаксис с (+))
        FOR (
            SELECT ac(ac%id : id
                    , ac.[MAIN_V_ID] : main_v_id
                    , gj.[WORKING_DATETIME] : WrkDate
                    , st.tm : AnswDate
                    , st.val : RESULT
                    , gj.[FILENAME] : FILENAME)
            IN ::[AC_FIN]
            , (SELECT gj(gj.[WORKING_DATETIME] : WORKING_DATETIME
                       , gj.[FILENAME] : FILENAME
                       , gj.[ACCOUNT] : ACCOUNT
                       , gj.[IN_FILE_HISTORY] : IN_FILE_HISTORY)
               IN ::[GNI_JOUR]
               WHERE coalesce(gj.[FILENAME],'SBC2') LIKE 'SBC2%' 
                  OR coalesce(gj.[FILENAME],'SFC2') LIKE 'SFC2%')
            , (SELECT DISTINCT st(st%collection : collection_id
                                , analytic(first_value(substr(st.[TO_FILENAME], 3, 1)), 'over (partition by [1] order by [2] desc)', st%collection, st.[IN_FILE_DATETIME]) : val
                                , analytic(first_value(st.[IN_FILE_DATETIME]), 'over (partition by [1] order by [2] desc)', st%collection, st.[IN_FILE_DATETIME]) : tm)
               IN ::[GNI_JOUR_HISTORY] collections
               WHERE st.[SOC_FUND] IS NULL 
                 AND st.[IN_FILE_DATETIME] < (P_DATE + 1))
            WHERE ac%id = gj.[ACCOUNT](true)
              AND gj.[IN_FILE_HISTORY] = st.collection_id(true)
        )
        LOOP
            EXIT;
        END LOOP;

        -- [+] ПРАВИЛЬНО (исправлено на LEFT JOIN)
        FOR (
            WITH gj(WORKING_DATETIME, FILENAME, ACCOUNT, IN_FILE_HISTORY) AS (
                SELECT gj(gj.[WORKING_DATETIME] : WORKING_DATETIME
                        , gj.[FILENAME] : FILENAME
                        , gj.[ACCOUNT] : ACCOUNT
                        , gj.[IN_FILE_HISTORY] : IN_FILE_HISTORY)
                IN ::[GNI_JOUR]
                WHERE (coalesce(gj.[FILENAME],'SBC2') LIKE 'SBC2%' 
                    OR coalesce(gj.[FILENAME],'SFC2') LIKE 'SFC2%')
            )
            , st(collection_id, val, tm) AS (
                SELECT DISTINCT st(st%collection : collection_id
                                 , analytic(first_value(substr(st.[TO_FILENAME], 3, 1)), 'over (partition by [1] order by [2] desc)', st%collection, st.[IN_FILE_DATETIME]) : val
                                 , analytic(first_value(st.[IN_FILE_DATETIME]), 'over (partition by [1] order by [2] desc)', st%collection, st.[IN_FILE_DATETIME]) : tm)
                IN ::[GNI_JOUR_HISTORY] collections
                WHERE st.[SOC_FUND] IS NULL 
                  AND st.[IN_FILE_DATETIME] < (P_DATE + 1)
            )
            SELECT ac(ac%id : id
                    , ac.[MAIN_V_ID] : main_v_id
                    , gj.[WORKING_DATETIME] : WrkDate
                    , st.tm : AnswDate
                    , st.val : RESULT)
            IN ::[AC_FIN]
            LEFT JOIN gj ON ac%id = gj.[ACCOUNT]
            LEFT JOIN st ON gj.[IN_FILE_HISTORY] = st.collection_id
        )
        LOOP
            EXIT;
        END LOOP;

        -- Прерываем выполнение, чтобы не вносить изменения в Базу
        RAISE; -- прерываем, чтобы не вносить изменение в Базу

    EXCEPTION WHEN OTHERS THEN
        debug_pipe(substr(utils.error_stack, 1, 4000), 0);
        ROLLBACK TO sp1;
        RAISE; -- прерываем, чтобы не вносить изменение в Базу
    END;
END;
