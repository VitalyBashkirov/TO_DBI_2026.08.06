```powershell
Stop-Transcript
Get-ChildItem $PSDir -File |
    Where-Object {
        $_.Length -eq 0 -and
        $_.Name -ne (Split-Path $log -Leaf) -and
        $_.LastWriteTime -lt (Get-Date).AddMinutes(-1)
    } |
    Remove-Item -Force -ErrorAction SilentlyContinue

```

## Генерация .ps1/.py (64_1, 64_4)
```powershell
# .ps1/.py с кириллицей -> BOM = $true
$encBom = New-Object System.Text.UTF8Encoding($true)
[System.IO.File]::WriteAllLines($gen, $L, $encBom)
$b = [System.IO.File]::ReadAllBytes($gen)
$isBom = ($b.Length -ge 3 -and $b[0] -eq 0xEF -and $b[1] -eq 0xBB -and $b[2] -eq 0xBF)
if (-not $isBom) { throw "BOM не записан — 64_4" }

# .md/.txt/.json/msgfile -> BOM = $false

[System.IO.File]::WriteAllText($p, $t, $encNoBom)
```
